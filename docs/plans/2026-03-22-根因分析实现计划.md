```python
from dataclasses import dataclass, field
from typing import Optional

@dataclass
class FaultAnalysisInput:
    """故障分析输入"""
    task_no: str
    title: str
    description: str
    task_src: str
    created_date: str
    finish_date: str
    product_module_id: Optional[int]
    product_version_id: Optional[int]

@dataclass
class ExistingFaultAnalysis:
    """现有故障复盘结论"""
    dev_catalog: str = ""           # 研发环节分类
    dev_catalog_detail: str = ""   # 研发环节细分类
    dev_reason: str = ""           # 研发原因
    dev_conclusion: str = ""       # 研发结论
    dev_improve_stage: str = ""    # 研发改进环节
    test_catalog: str = ""          # 测试环节分类
    test_catalog_detail: str = ""   # 测试环节细分类
    test_reason: str = ""          # 测试原因
    test_conclusion: str = ""       # 测试结论
    test_improve_stage: str = ""   # 测试改进环节

@dataclass
class RootCause:
    """深层根因"""
    layer: str                      # 层面：设计/编码/测试/流程/知识管理
    root_cause: str                # 根因描述
    why_reason: str                 # 追问为什么
    evidence: str                   # 支撑证据

@dataclass
class ActionableImprovement:
    """可落地改进措施"""
    type: str                       # 直接改进/流程改进
    action: str                     # 具体措施
    owner: str                      # 责任方
    priority: str                   # 高/中/低

@dataclass
class RootCauseAnalysisResult:
    """根因分析结果"""
    problem_category: str                            # 问题分类
    initial_cause: str                               # 初步归因
    deep_root_causes: list[RootCause] = field(default_factory=list)
    actionable_improvements: list[ActionableImprovement] = field(default_factory=list)
    checklist_recommendations: list[str] = field(default_factory=list)
```

```python
from dataclasses import asdict
from src.analysis.root_cause.models import (
    FaultAnalysisInput, ExistingFaultAnalysis, RootCauseAnalysisResult
)
from src.analysis.root_cause.prompts import ROOT_CAUSE_ANALYSIS_PROMPT

class RootCauseAnalyzer:
    """根因分析器"""

    def __init__(self, llm_client):
        self.llm_client = llm_client

    def _build_prompt(self, fault_input: FaultAnalysisInput,
                      existing_analysis: ExistingFaultAnalysis) -> str:
        """构建Prompt"""
        return ROOT_CAUSE_ANALYSIS_PROMPT.format(
            task_no=fault_input.task_no,
            title=fault_input.title,
            description=fault_input.description,
            task_src=fault_input.task_src,
            created_date=fault_input.created_date,
            finish_date=fault_input.finish_date,
            dev_catalog=existing_analysis.dev_catalog,
            dev_catalog_detail=existing_analysis.dev_catalog_detail,
            dev_reason=existing_analysis.dev_reason,
            dev_conclusion=existing_analysis.dev_conclusion,
            dev_improve_stage=existing_analysis.dev_improve_stage,
            test_catalog=existing_analysis.test_catalog,
            test_catalog_detail=existing_analysis.test_catalog_detail,
            test_reason=existing_analysis.test_reason,
            test_conclusion=existing_analysis.test_conclusion,
            test_improve_stage=existing_analysis.test_improve_stage,
        )

    async def analyze(self, fault_input: FaultAnalysisInput,
                      existing_analysis: ExistingFaultAnalysis) -> RootCauseAnalysisResult:
        """执行根因分析"""
        prompt = self._build_prompt(fault_input, existing_analysis)
        response = await self.llm_client.generate(prompt)

        # 解析JSON响应
        import json
        result_data = json.loads(response)

        # 转换为数据类
        return RootCauseAnalysisResult(**result_data)
```

## 附录：根因分类体系

### 6.1 大类体系

| 大类 | 说明 |
|-----|------|
| 研发环节 | 代码编写、逻辑设计相关问题 |
| 测试设计类 | 测试用例设计、覆盖相关问题 |
| 需求分析类 | 需求描述、确认相关问题 |
| 流程管理类 | 评审、发布、变更流程相关问题 |
| 知识传承类 | 经验总结、培训、文档相关问题 |

### 6.2 细分类体系（参考现有故障复盘数据）

**研发环节**：
- 正常场景遗漏
- 异常场景遗漏
- 边界条件未覆盖
- 代码逻辑错误
- 防御性编程不足
- 异常处理不完善

**测试设计类**：
- 关联场景考虑不全
- 测试路径覆盖不全
- 测试数据准备不充分
- 测试用例设计方法缺失
- checklist不完善

### 6.3 根因-改进措施映射库

| 根因模式 | 对应改进措施 |
|---------|------------|
| 状态转换场景遗漏 | 代码走查checklist增加"状态机覆盖验证" |
| 接口契约不清晰 | 设计规范要求"接口双方签名确认" |
| 边界条件未覆盖 | 测试用例规范增加"边界值分析"章节 |
| 历史问题未复用 | 建立"同类问题复盘checklist" |
