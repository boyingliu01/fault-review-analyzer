## 三、引入环节聚类分析

### 3.1 分析代码

```python
import pandas as pd
import numpy as np
from sklearn.cluster import KMeans
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics import silhouette_score

# 读取数据
df = pd.read_excel('国际中心2025年关闭的已集中复盘的泄露缺陷单.xlsx')

# 引入环节特征
intro_cols = ['引入环节分类', '引入环节', '引入原因', '复盘结论', '引入改进环节', '引入改进措施']
intro_text = df[intro_cols].fillna('').apply(lambda x: ' '.join(x), axis=1)

# TF-IDF向量化
tfidf = TfidfVectorizer(max_features=500)
intro_features = tfidf.fit_transform(intro_text)

# 确定最佳簇数
silhouette_scores = []
for k in range(3, 8):
    kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
    labels = kmeans.fit_predict(intro_features)
    score = silhouette_score(intro_features, labels)
    silhouette_scores.append(score)

best_k = 3 + silhouette_scores.index(max(silhouette_scores))
print(f"最佳簇数: {best_k}")

# 聚类
kmeans = KMeans(n_clusters=best_k, random_state=42, n_init=10)
df['intro_cluster'] = kmeans.fit_predict(intro_features)

# 分析结果
for i in range(best_k):
    cluster_data = df[df['intro_cluster'] == i]
    print(f"\n簇 {i+1}: {len(cluster_data)}条 ({len(cluster_data)/len(df)*100:.1f}%)")
    print("引入环节Top3:", cluster_data['引入环节'].value_counts().head(3))
    print("引入原因Top3:", cluster_data['引入原因'].value_counts().head(3))
    print("改进措施Top3:", cluster_data['引入改进措施'].value_counts().head(3))
```
