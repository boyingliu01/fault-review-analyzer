## 3. 分析实施步骤

### 3.1 数据预处理

```python
# 1. 读取数据
df = pd.read_excel('国际中心2025年关闭的已集中复盘的泄露缺陷单.xlsx')

# 2. 数据清洗
df = df.dropna(subset=['引入环节', '泄露原因'])

# 3. 特征工程
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.feature_extraction.text import TfidfVectorizer

# 文本字段向量化
tfidf = TfidfVectorizer(max_features=1000, stop_words='english')
text_features = tfidf.fit_transform(df['引入改进措施'].fillna('') + ' ' + df['泄露改进措施'].fillna(''))

# 分类字段编码
le = LabelEncoder()
categorical_features = []
for col in ['引入环节', '复盘结论', '引入原因', '引入改进环节', '泄露原因', '泄漏根因环节', '泄漏根因类型', '泄露改进环节']:
    if col in df.columns:
        df[col + '_encoded'] = le.fit_transform(df[col].astype(str))
        categorical_features.append(col + '_encoded')

# 合并特征
from scipy.sparse import hstack
X = hstack([text_features, df[categorical_features]])
```

### 3.2 聚类分析

```python
# 1. 确定最佳簇数
from sklearn.cluster import KMeans
import matplotlib.pyplot as plt

inertia = []
for k in range(2, 10):
    kmeans = KMeans(n_clusters=k, random_state=42)
    kmeans.fit(X)
    inertia.append(kmeans.inertia_)

plt.plot(range(2, 10), inertia)
plt.xlabel('簇数')
plt.ylabel('惯性')
plt.title('肘部法则')
plt.show()

# 2. 执行K-means聚类
kmeans = KMeans(n_clusters=5, random_state=42)
df['cluster'] = kmeans.fit_predict(X)

# 3. 分析聚类结果
for i in range(5):
    cluster_data = df[df['cluster'] == i]
    print(f"\n簇 {i+1}:")
    print(f"样本数量: {len(cluster_data)}")
    print("引入环节Top3:", Counter(cluster_data['引入环节']).most_common(3))
    print("泄露原因Top3:", Counter(cluster_data['泄露原因']).most_common(3))
```

### 3.3 结果可视化

```python
# 使用PCA降维可视化
from sklearn.decomposition import PCA

pca = PCA(n_components=2)
x_pca = pca.fit_transform(X.toarray())

plt.figure(figsize=(10, 8))
for i in range(5):
    plt.scatter(x_pca[df['cluster'] == i, 0], x_pca[df['cluster'] == i, 1], label=f'簇 {i+1}')
plt.legend()
plt.title('聚类结果可视化')
plt.show()
```
